         IF (NOT @Quantidade_Entregue_Pedido_Venda IS NULL)
         BEGIN
             INSERT INTO Movimento_Prod_Serv_Pedido_Venda_Entrega
             VALUES (
             @ID
             ,@Quantidade_Entregue_Pedido_Venda
             )

             SET Pedidos_Venda = Pedidos_Venda - @Quantidade_Entregue_Pedido_Venda
